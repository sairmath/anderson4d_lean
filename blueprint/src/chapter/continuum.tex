\chapter{Layer L2: continuum estimates}\label{ch:continuum}

Deterministic Fourier core, the torus Green's function, singular
convolutions, and the discretization that feeds the lattice machine.

\begin{definition}[Symmetry class \(\mathcal{E}\); paper Def.~2.1]\label{D-E}
  \lean{Anderson4D.MemEClassR4, Anderson4D.MemEClassT4,
    Anderson4D.SmoothCutoff}\leanok
  Functions on \(\mathbb{T}^4\) (resp.\ \(\mathbb{R}^4\)) invariant under
  coordinate permutations and per-coordinate sign flips (hyperoctahedral
  \(B_4\)). The formalization uses this single-slot interpretation.
\end{definition}

\begin{definition}[Deterministic Fourier core]\label{I-torus}
  \lean{Anderson4D.torusFourierBasis,
    Anderson4D.hasSum_torusFourier_series_L2,
    Anderson4D.tsum_sq_torusFourierCoeff,
    Anderson4D.torusL2MultiplierCLM,
    Anderson4D.paperKernelCoeff_eq_volume_sq_smul_normalized}\leanok
  Characters \(e_k\), kernel coefficients (3.23)
  \(\widehat K(\alpha,\beta) = \langle e_{-\alpha}, K e_\beta\rangle\),
  Plancherel on the product torus, multiplier calculus, and the
  normalization ledger dictionary.
\end{definition}

\begin{definition}[Green's function; paper (4.1)]\label{I-green}
  \uses{I-torus,D-E}
  \lean{Anderson4D.heatKernelT4, Anderson4D.greenFn,
    Anderson4D.greenKernel, Anderson4D.fourPointH,
    Anderson4D.paperFourierCoeff_greenFn,
    Anderson4D.greenFn_memE,
    Anderson4D.abs_greenLocalLift_singular,
    Anderson4D.abs_greenLocalGrad_singular,
    Anderson4D.abs_greenLocalHess_singular,
    Anderson4D.abs_greenLocalImprovedRemainder_le}\leanok
  \(G := \int_0^\infty e^{-t}\,\Theta(t,\cdot)\,\mathrm{d}t\) with
  \(\Theta\) the periodized Gaussian (Bessel-potential representation;
  a Fourier \texttt{tsum} is ill-defined since
  \(\langle k\rangle^{-2}\notin\ell^1\cup\ell^2\) and
  \(G \notin L^2\)). Lemmas: \(\widehat G(k) = \langle k\rangle^{-2}\);
  \(G \in \mathcal{E}\); \(\lvert\nabla^k G(z)\rvert \lesssim
  \lvert z\rvert^{-2-k}\) (\(k \le 2\)); the improved near-diagonal bound
  for \(G - \frac{1}{4\pi^2\lvert z\rvert^2}\).
\end{definition}
